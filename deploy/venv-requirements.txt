# Runtime Python wheels for the exec venv (/opt/venv) that the API spawns for
# every `python3 <script>.py`. These are the REAL, pip-installable dependencies
# the server-side scripts need — kept separate from ../baja-apps/py/requirements.txt,
# which is an import-NAME list (many entries are stdlib or non-PyPI, e.g. os/json/PIL)
# and therefore cannot be `pip install -r`'d wholesale.
#
# Minimal by design: add a wheel here only when a server-invoked script imports it.
#   numpy        -> off-target search   (py/sequence/offtarget/search.py, build-index.py)
#   requests     -> Claude gene resolver (py/sequence/prompt-to-transcript.py, SNP info)
#   pandas       -> djPrimer            (py/djprimer-lib/djprimer/service.py)
#   scikit-learn -> ML scripts (pulls scipy+joblib); joblib loads saved models
#   joblib       -> predict_solubility  (py/bio/protein/predict_solubility.py)
#   pyBigWig     -> RNA-seq rendering    (py/baja/bigwig/view-bigwig.py)
#   pyfaidx      -> bajasplice.genome    (py/bajasplice-lib/bajasplice/genome.py)
numpy>=1.26
requests>=2.25
pandas>=2.0
scikit-learn>=1.3
joblib>=1.3
pyBigWig>=0.3.22
pyfaidx>=0.7
#
# Editable local packages (ion, bajasplice, bajaclip, djprimer) are NOT wheels and
# live under baja-apps/py/*-lib, so they can't be pinned here. They are installed
# by deploy.sh AFTER the apps are synced (see "install exec-venv local libs").
